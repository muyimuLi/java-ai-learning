# 接入QuestionAnswerAdvisors
来自`spring-ai-vector-store-advisor`包

## 示例代码
pom.xml：
```xml
<dependency>
    <groupId>org.springframework.ai</groupId>
    <artifactId>spring-ai-vector-store-advisor</artifactId>
</dependency>
```

ChatClientConfig：
```java
@Configuration
public class ChatClientConfig {
    @Bean
    public ChatClient chatClient(ChatClient.Builder builder) {
        return builder.defaultAdvisors(new MyLoggerAdvisor()).build();
    }
}
```

DocumentIntegrationService：
```java
@Service
public class DocumentIntegrationService {
    private static final String ANSWER_SYSTEM_PROMPT = """
            你是企业知识库问答助手。严格遵守以下规则：
            
            1. 只依据【资料】中的内容回答，不要使用你自己的知识。
            2. 每个结论后面必须用 [编号] 标注来源，例如：研发投入为 2.16 亿元[1]。
            3. 如果【资料】中找不到答案，直接回答"资料中未提及"，不要推测、不要补充。
            4. 如果资料之间互相矛盾，指出矛盾点，不要自行取舍。
            5. 不要写"根据资料""如上所述"这类套话，直接给结论。
            """;

    @Autowired
    private ChatClient chatClient;

    @Autowired
    private PgVectorStore pgVectorStore;

    public Map<String, Object> retrieveDocuments(String userMessage, Integer mode) {
        return switch (mode) {
            case 1 -> retrieveByQuestionAnswerAdvisor(userMessage);
            case 2 -> retrieveByVectorStoreDocumentRetriever(userMessage);
            case 3 -> retrieveByVectorStoreRetriever(userMessage);
            default -> throw new IllegalArgumentException("not supported mode");
        };
    }

    private Map<String, Object> retrieveByQuestionAnswerAdvisor(String userMessage) {
        ChatClientResponse chatClientResponse = chatClient.prompt(userMessage)
                .advisors(QuestionAnswerAdvisor.builder(pgVectorStore)
                        .searchRequest(SearchRequest.builder()
                                .topK(5)
                                .similarityThreshold(0.3)
                                .build())
                        .build())
                .call()
                .chatClientResponse();
        String answer = chatClientResponse.chatResponse().getResult().getOutput().getText();
        List<Map<String, Object>> source = new ArrayList<>();
        List<Document> documents = (List<Document>) chatClientResponse.context().get(QuestionAnswerAdvisor.RETRIEVED_DOCUMENTS);
        if (documents != null && !documents.isEmpty()) {
            source = documents.stream().map(document -> Map.of("index", document.getId(),
                            "file", document.getMetadata().get("filename"),
                            "chunk", document.getMetadata().get("chunk_index"),
                            "text", document.getText()))
                    .toList();
        }
        return Map.of("answer", answer, "source", source);
    }

    private Map<String, Object> retrieveByVectorStoreDocumentRetriever(String userMessage) {
        VectorStoreDocumentRetriever vectorStoreDocumentRetriever = VectorStoreDocumentRetriever.builder()
                .topK(5)
                .similarityThreshold(0.3)
                .vectorStore(pgVectorStore)
                .build();
        Query query = new Query(userMessage);
        List<Document> documents = vectorStoreDocumentRetriever.retrieve(query);
        return buildAnswerResponse(documents, userMessage);
    }

    private Map<String, Object> retrieveByVectorStoreRetriever(String userMessage) {
        List<Document> documents = pgVectorStore.similaritySearch(SearchRequest.builder().query(userMessage).build());
        return buildAnswerResponse(documents, userMessage);
    }

    private Map<String, Object> buildAnswerResponse(List<Document> documents, String userMessage) {
        String answer = "";
        List<Map<String, Object>> source = new ArrayList<>();
        if (documents != null && !documents.isEmpty()) {
            StringBuilder context = new StringBuilder();
            for (int i = 0; i < documents.size(); i++) {
                context.append("[").append(i + 1).append("] ")
                        .append(documents.get(i).getText())
                        .append("\n\n");
            }

            answer = chatClient.prompt()
                    .system(ANSWER_SYSTEM_PROMPT)
                    .user(u -> u.text("""
                                    资料：
                                    {context}
                                    
                                    问题：{question}
                                    """)
                            .param("context", context.toString())
                            .param("question", userMessage))
                    .call()
                    .content();

            source = IntStream.range(0, documents.size())
                    .mapToObj(i -> Map.of(
                            "index", i + 1,
                            "chunk_index", documents.get(i).getMetadata().getOrDefault("chunk_index", "未知"),
                            "preview", documents.get(i).getText().substring(
                                    0, Math.min(60, documents.get(i).getText().length()))))
                    .toList();
        }

        return Map.of("answer", answer, "source", source);
    }

}
```

DocumentIntegrationController：
```java
@RestController
@RequestMapping("/ai/embedding")
public class DocumentIntegrationController {
    @Autowired
    private DocumentIntegrationService documentIntegrationService;

    @GetMapping("/retrieve")
    public Object retrieveDocuments(@RequestParam("userMessage") String userMessage, @RequestParam("mode") Integer mode) {
        return documentIntegrationService.retrieveDocuments(userMessage, mode);
    }

}
```
