# 接入QuestionAnswerAdvisors
来自`spring-ai-vector-store-advisor`包

## 示例代码
pom.xml：
```xml
<dependency>
    <groupId>org.springframework.ai</groupId>
    <artifactId>spring-ai-vector-store-advisor</artifactId>
</dependency>
```

ChatClientConfig：
```java
@Configuration
public class ChatClientConfig {
    @Autowired
    private PgVectorStore pgVectorStore;

    @Bean
    public ChatClient chatClient(ChatClient.Builder builder) {
        return builder.defaultAdvisors(QuestionAnswerAdvisor.builder(pgVectorStore).build()).build();
    }
}
```

DocumentIntegrationService：
```java
@Service
public class DocumentIntegrationService {
    @Autowired
    private ChatClient chatClient;

    @Autowired
    private PgVectorStore pgVectorStore;

    public Map<String, Object> retrieveDocuments(String userMessage, Integer mode) {
        return switch (mode) {
            case 1 -> retrieveByQuestionAnswerAdvisor(userMessage);
            case 2 -> retrieveByVectorStoreDocumentRetriever(userMessage);
            case 3 -> retrieveByVectorStoreRetriever(userMessage);
            default -> throw new IllegalArgumentException("not supported mode");
        };
    }

    private Map<String, Object> retrieveByQuestionAnswerAdvisor(String userMessage) {
        ChatClientResponse chatClientResponse = chatClient.prompt(userMessage)
                .call()
                .chatClientResponse();
        String response = chatClientResponse.chatResponse().getResult().getOutput().getText();
        List<Document> documents = (List<Document>) chatClientResponse.context()
                .get(QuestionAnswerAdvisor.RETRIEVED_DOCUMENTS);
        return Map.of("response", response, "documents", documents);
    }

    private Map<String, Object> retrieveByVectorStoreDocumentRetriever(String userMessage) {
        VectorStoreDocumentRetriever vectorStoreDocumentRetriever = VectorStoreDocumentRetriever.builder()
                .vectorStore(pgVectorStore)
                .build();
        Query query = new Query(userMessage);
        List<Document> documents = vectorStoreDocumentRetriever.retrieve(query);
        List<String> response = documents.stream().map(Document::getText).collect(Collectors.toUnmodifiableList());
        return Map.of("response", response, "documents", documents);
    }

    private Map<String, Object> retrieveByVectorStoreRetriever(String userMessage) {
        List<Document> documents = pgVectorStore.similaritySearch(SearchRequest.builder()
                .query(userMessage)
                .build());
        List<String> response = documents.stream().map(Document::getText).collect(Collectors.toUnmodifiableList());
        return Map.of("response", response, "documents", documents);
    }

}
```

DocumentIntegrationController：
```java
@RestController
@RequestMapping("/ai/embedding")
public class DocumentIntegrationController {
    @Autowired
    private DocumentIntegrationService documentIntegrationService;

    @GetMapping("/retrieve")
    public Object retrieveDocuments(@RequestParam("userMessage") String userMessage, @RequestParam("mode") Integer mode) {
        return documentIntegrationService.retrieveDocuments(userMessage, mode);
    }

}
```
